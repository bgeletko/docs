import java.awt.Color; /** * ITSC 1213 Module 12 * The Circle class represents a circle with a radius and color. * Includes methods for Circle information and calculation of area. * * @author nanajjar * @version Spring 25 */ public class Circle { // private instance variable, not accessible from outside this class private double radius; private Color color; private double area; /** * The default constructor creating a Circle with default values of radius 1.0 and orange color. */ public Circle() { radius = 1.0; this.color = new Color(255,119,65); } /** * Creates Circle with specific radius and color * * @param r for circle's radius * @param clr for circle's color */ public Circle(double r, Color clr) { radius = r; color = clr; } /** * Returns circle's radius * @return radius of the circle */ public double getRadius() { return radius; } /** * Returns circle's color * @return color of the circle */ public Color getColor() { return color; } /** * Calculates and Returns circle's area * @return area of the circle */ public double getArea() { calculateArea(); return area; } /** * Private method to calculate area of the circle * Sets answer equal to area variable */ private void calculateArea() { area = radius * radius * Math.PI; } }
